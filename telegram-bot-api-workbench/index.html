Minimal form-encoded request wrapper

async function telegram(token, method, params = {}, { signal } = {}) {
  if (!/^\d+:[A-Za-z0-9_-]{20,}$/.test(token)) {
    throw new Error("Valid bot token required");
  }

  const body = new URLSearchParams();
  for (const [key, value] of Object.entries(params)) {
    if (value === undefined || value === null || value === "") continue;
    body.set(key, typeof value === "object" ? JSON.stringify(value) : String(value));
  }

  const response = await fetch(
    `https://api.telegram.org/bot${token}/${method}`,
    { method: "POST", body, signal }
  );
  const data = await response.json();
  if (!response.ok || !data.ok) {
    const retry = data.parameters?.retry_after;
    throw new Error(`${data.description || `HTTP ${response.status}`}${retry ? `; retry after ${retry}s` : ""}`);
  }
  return data.result;
}

Long polling with offset advancement

let offset;
const controller = new AbortController();

async function poll() {
  while (!controller.signal.aborted) {
    const updates = await telegram(token, "getUpdates", {
      offset,
      timeout: 25,
      allowed_updates: ["message", "callback_query"]
    }, { signal: controller.signal });

    for (const update of updates) {
      console.log(update);
      offset = update.update_id + 1;
    }
  }
}

poll().catch(error => {
  if (error.name !== "AbortError") console.error(error);
});

Multipart upload

async function telegramUpload(token, method, fields, fileField, file) {
  const form = new FormData();
  for (const [key, value] of Object.entries(fields)) {
    form.append(key, typeof value === "object" ? JSON.stringify(value) : String(value));
  }
  form.append(fileField, file);

  const response = await fetch(
    `https://api.telegram.org/bot${token}/${method}`,
    { method: "POST", body: form }
  );
  const data = await response.json();
  if (!response.ok || !data.ok) throw new Error(data.description || `HTTP ${response.status}`);
  return data.result;
}
